import re
from pathlib import Path

pattern = re.compile(
    r'api[_-]?key|token|secret|password|bearer|authorization|private[_-]?key',
    re.I
)

p = Path("AGENTS.md")

for n, line in enumerate(p.read_text(errors="ignore").splitlines(), 1):
    matches = sorted(set(m.lower() for m in pattern.findall(line)))
    if matches:
        print(f"Line {n}: {', '.join(matches)}")
